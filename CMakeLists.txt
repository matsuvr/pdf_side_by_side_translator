cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(cmake/Product.cmake)
project(OfflinePdfTranslator VERSION ${OPT_VERSION} LANGUAGES C CXX)
if(APPLE)
    enable_language(OBJC OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_SKIP_RPATH TRUE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(OPT_DEPENDENCIES "${PROJECT_SOURCE_DIR}/.work/src" CACHE PATH "Prepared official sources")
set(OPT_TARGET "macos-arm64-release" CACHE STRING "Dependency configuration")
set(OPT_PDFIUM_MANIFEST "${PROJECT_SOURCE_DIR}/.work/pdfium/${OPT_TARGET}/pdfium-build-manifest.json" CACHE FILEPATH "GN-resolved link closure")
option(OPT_BUILD_PDF_PROBE "Build the CMake PDFium link consumer" ON)
option(OPT_BUILD_GUI_PROBE "Build the combined native GUI dependency probe" OFF)

foreach(dep llama.cpp SQLite)
    if(NOT EXISTS "${OPT_DEPENDENCIES}/${dep}")
        message(FATAL_ERROR "Missing prepared ${dep}: run scripts/bootstrap.py")
    endif()
endforeach()

set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU ON CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE ${APPLE} CACHE BOOL "" FORCE)
if(APPLE AND CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
    set(GGML_METAL ON CACHE BOOL "" FORCE)
    set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
else()
    set(GGML_METAL OFF CACHE BOOL "" FORCE)
endif()
foreach(backend CUDA VULKAN RPC HIP SYCL OPENCL OPENVINO WEBGPU ZDNN VIRTGPU MUSA HEXAGON)
    set(GGML_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(isa SSE42 AVX AVX2 AVX_VNNI BMI2 AVX512 AVX512_VBMI AVX512_VNNI AVX512_BF16 FMA F16C AMX_TILE AMX_INT8 AMX_BF16)
    set(GGML_${isa} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(component COMMON TESTS TOOLS EXAMPLES SERVER APP UI MTMD)
    set(LLAMA_BUILD_${component} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${OPT_DEPENDENCIES}/llama.cpp" third_party/llama EXCLUDE_FROM_ALL)
add_library(opt_sqlite STATIC "${OPT_DEPENDENCIES}/SQLite/sqlite3.c")
target_include_directories(opt_sqlite PUBLIC "${OPT_DEPENDENCIES}/SQLite")
target_compile_definitions(opt_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION SQLITE_THREADSAFE=1 SQLITE_DQS=0)
add_library(opt_inference STATIC src/inference/engine.cpp)
target_include_directories(opt_inference PUBLIC src)
target_link_libraries(opt_inference PRIVATE llama)

include(CTest)
add_executable(inference_probe tests/integration/inference_probe.cpp)
target_link_libraries(inference_probe PRIVATE opt_inference)
add_executable(utf8_test tests/unit/utf8_test.cpp)
target_include_directories(utf8_test PRIVATE src)
add_test(NAME utf8_stream COMMAND utf8_test)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME binary_dependency_audit COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/unit/test_dependency_audit.py")
endif()
if(OPT_BUILD_PDF_PROBE)
    include(cmake/PDFium.cmake)
    add_executable(pdf_probe tests/integration/pdf_probe.cpp)
    target_link_libraries(pdf_probe PRIVATE PDFium::PDFium)
    add_test(NAME pdf_render_extract COMMAND pdf_probe "${PROJECT_SOURCE_DIR}/tests/fixtures/P01-simple.pdf")
endif()

if(OPT_BUILD_GUI_PROBE)
    if(NOT OPT_BUILD_PDF_PROBE)
        message(FATAL_ERROR "The combined GUI probe requires the validated PDFium import")
    endif()
    set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
    if(WIN32)
        set(wxBUILD_USE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
    endif()
    set(wxBUILD_INSTALL OFF CACHE BOOL "" FORCE)
    set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
    set(wxUSE_LIBTIFF builtin CACHE STRING "" FORCE)
    set(wxUSE_LIBJPEG builtin CACHE STRING "" FORCE)
    set(wxUSE_LIBPNG builtin CACHE STRING "" FORCE)
    set(wxUSE_LIBWEBP builtin CACHE STRING "" FORCE)
    set(wxUSE_ZLIB builtin CACHE STRING "" FORCE)
    set(wxUSE_EXPAT builtin CACHE STRING "" FORCE)
    set(wxUSE_LIBLZMA OFF CACHE BOOL "" FORCE)
    set(wxUSE_OPENGL OFF CACHE BOOL "" FORCE)
    set(wxUSE_MEDIACTRL OFF CACHE BOOL "" FORCE)
    set(wxUSE_SOCKETS OFF CACHE BOOL "" FORCE)
    set(wxUSE_URL OFF CACHE BOOL "" FORCE)
    set(wxUSE_PROTOCOL OFF CACHE BOOL "" FORCE)
    set(wxUSE_PROTOCOL_HTTP OFF CACHE BOOL "" FORCE)
    set(wxUSE_PROTOCOL_FTP OFF CACHE BOOL "" FORCE)
    set(wxUSE_PROTOCOL_FILE OFF CACHE BOOL "" FORCE)
    set(wxUSE_FS_INET OFF CACHE BOOL "" FORCE)
    set(wxUSE_WEBREQUEST OFF CACHE BOOL "" FORCE)
    set(wxUSE_IPC OFF CACHE BOOL "" FORCE)
    set(wxUSE_FSWATCHER OFF CACHE BOOL "" FORCE)
    add_subdirectory("${OPT_DEPENDENCIES}/wxWidgets" third_party/wx EXCLUDE_FROM_ALL)
    set(OPT_FONT_RESOURCE "${PROJECT_SOURCE_DIR}/.work/resources/NotoSansCJKjp-Regular.otf")
    if(APPLE)
        enable_language(OBJCXX)
        add_library(opt_platform STATIC src/platform/macos/font.mm)
        target_link_libraries(opt_platform PRIVATE "-framework CoreText" "-framework Foundation")
        set_source_files_properties("${OPT_FONT_RESOURCE}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
        add_executable(dependency_probe MACOSX_BUNDLE tests/integration/gui_probe.cpp "${OPT_FONT_RESOURCE}")
        set_target_properties(dependency_probe PROPERTIES
            OUTPUT_NAME "${OPT_PRODUCT_NAME} Probe"
            MACOSX_BUNDLE_GUI_IDENTIFIER "${OPT_BUNDLE_ID}.probe")
    elseif(WIN32)
        enable_language(RC)
        add_library(opt_platform STATIC src/platform/windows/font.cpp)
        target_link_libraries(opt_platform PRIVATE gdi32)
        configure_file(resources/windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/windows.rc" @ONLY)
        add_executable(dependency_probe WIN32 tests/integration/gui_probe.cpp "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
    else()
        message(FATAL_ERROR "Only Windows and macOS distribution targets are supported")
    endif()
    target_include_directories(opt_platform PUBLIC src)
    target_link_libraries(dependency_probe PRIVATE opt_inference opt_sqlite opt_platform PDFium::PDFium wx::core wx::base)
endif()
